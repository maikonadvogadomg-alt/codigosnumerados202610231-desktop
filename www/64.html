⌨️ Terminal do PC — Conectar ao Banco

No app Electron (PC), o ícone 🖥️ abre um terminal real. Você pode instalar pacotes, rodar scripts Node.js e conectar ao banco diretamente.

1️⃣ Instalar pacote do banco

Cole no terminal e pressione Enter:

npm install @neondatabase/serverless

2️⃣ Criar arquivo .env

Crie o arquivo .env na raiz do projeto:

DATABASE_URL=postgresql://usuario:senha@ep-xxx.us-east-2.aws.neon.tech/banco?sslmode=require

3️⃣ Testar a conexão

Crie o arquivo teste-db.js:

require('dotenv').config();
const { Pool } = require('pg');

const pool = new Pool({ connectionString: process.env.DATABASE_URL });

async function testar() {
  try {
    const res = await pool.query('SELECT NOW() AS agora, version() AS versao');
    console.log('✅ CONECTADO!');
    console.log('Data/hora:', res.rows[0].agora);
  } catch (err) {
    console.error('❌ ERRO:', err.message);
  } finally {
    await pool.end();
  }
}

testar();
